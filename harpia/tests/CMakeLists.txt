# Separate moc objects retain archive granularity: a test must not pull in every
# UI class (including recording/OBS integrations) merely to use one widget.
list(APPEND EDITOR_HEADERS "${H}/core/ShareExporter.hpp" "${H}/core/GlobalHotkeys.hpp")
foreach(header IN LISTS EDITOR_HEADERS)
  file(READ "${header}" contents)
  if(contents MATCHES "Q_OBJECT")
    file(RELATIVE_PATH relative "${H}" "${header}")
    string(REPLACE "/" "_" label "${relative}")
    set(output "${CMAKE_CURRENT_BINARY_DIR}/moc_${label}.cpp")
    add_custom_command(OUTPUT "${output}" COMMAND Qt6::moc "-I${H}" "${header}" -o "${output}" DEPENDS "${header}" VERBATIM)
    target_sources(harpia-test-editor PRIVATE "${output}")
  endif()
endforeach()

function(register_harpia_test name)
  target_sources("${name}_test" PRIVATE support/test_settings.cpp)
  target_compile_definitions("${name}_test" PRIVATE HARPIA_TEST_SETTINGS=1)
  add_dependencies(harpia-tests "${name}_test")
  add_test(NAME "${name}" COMMAND "${name}_test" "${MEDIA}")
  set_tests_properties("${name}" PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;HARPIA_TEST_MEDIA=${MEDIA}")
endfunction()

    add_test(NAME "${name}" COMMAND "${name}_test")

      set_tests_properties("${name}" PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
